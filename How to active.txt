e tbm Substitua a sua função image() pelo bloco abaixo:
/**
 * Carrega gráficos priorizando o formato .ctp otimizado para mobile.
 * Caso o arquivo .ctp não exista na pasta, faz o fallback para o .png tradicional.
 */
public static function image(key:String, ?library:String):FlxGraphic 
{
    // 1. Tenta carregar a textura binária otimizada em .ctp
    var ctpPath:String = getPath('images/' + key + '.ctp', IMAGE, library);
    if (Assets.exists(ctpPath)) 
    {
        var ctpGraphic:FlxGraphic = CTPGraphic.fromFile(ctpPath);
        if (ctpGraphic != null) return ctpGraphic;
    }

    // 2. Fallback: Carrega o PNG original caso o arquivo .ctp não exista
    var pngPath:String = getPath('images/' + key + '.png', IMAGE, library);
    if (Assets.exists(pngPath)) 
    {
        return FlxG.bitmap.add(pngPath, false, pngPath);
    }

    trace('[Paths] AVISO: Imagem não encontrada para key: ' + key);
    return null;
}
